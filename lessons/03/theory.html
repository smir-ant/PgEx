Оператор != (или <>) проверяет на неравенство.

SELECT * FROM employees WHERE department_id != 1;

Оператор LIKE используется для поиска по шаблону. % — любая строка, _ — один символ.

SELECT * FROM employees WHERE name LIKE 'J%';
SELECT * FROM employees WHERE email LIKE '%example.com';

NOT IN и NOT BETWEEN исключают значения из списка или диапазона.

SELECT * FROM employees WHERE department_id NOT IN (1, 2);
SELECT * FROM employees WHERE salary NOT BETWEEN 50000 AND 100000;
